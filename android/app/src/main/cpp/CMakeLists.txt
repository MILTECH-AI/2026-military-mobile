cmake_minimum_required(VERSION 3.22)
project(maintenance_llama LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
foreach(flag LLAMA_BUILD_COMMON LLAMA_BUILD_TESTS LLAMA_BUILD_TOOLS LLAMA_BUILD_EXAMPLES LLAMA_BUILD_SERVER LLAMA_BUILD_APP LLAMA_BUILD_UI LLAMA_OPENSSL LLAMA_SUBPROCESS GGML_NATIVE GGML_OPENMP GGML_LLAMAFILE GGML_CPU_KLEIDIAI)
  set(${flag} OFF CACHE BOOL "" FORCE)
endforeach()
if(NOT DEFINED LLAMA_SOURCE_DIR)
  message(FATAL_ERROR "Provision pinned llama.cpp source on the PC first")
endif()
add_subdirectory("${LLAMA_SOURCE_DIR}" llama EXCLUDE_FROM_ALL)
add_library(maintenance_llama SHARED llama_bridge.cpp)
target_link_libraries(maintenance_llama PRIVATE llama android log)
target_compile_options(maintenance_llama PRIVATE -fvisibility=hidden)
